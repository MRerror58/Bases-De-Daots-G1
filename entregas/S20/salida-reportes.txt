== Reporte 1. Los libros más prestados (los 3 primeros, con empates) ==
SELECT b.titulo, b.autores, count(p.id_prestamo) AS prestamos
FROM   libro_detalle b
LEFT JOIN ejemplar e ON e.isbn = b.isbn
LEFT JOIN prestamo p ON p.id_ejemplar = e.id_ejemplar
WHERE  b.activo
GROUP BY b.isbn, b.titulo, b.autores
ORDER BY prestamos DESC, b.titulo
FETCH FIRST 3 ROWS WITH TIES;

== Reporte 2. El o los lectores que más libros han leído (préstamos devueltos) ==
WITH leidos AS (
    SELECT id_lector, count(*) AS devueltos
    FROM   prestamo
    WHERE  fecha_devolucion IS NOT NULL
    GROUP BY id_lector
)
SELECT l.documento, l.nombres, l.apellidos, x.devueltos
FROM   leidos x
JOIN   lector l ON l.id_lector = x.id_lector
WHERE  x.devueltos = (SELECT max(devueltos) FROM leidos)
ORDER BY l.apellidos;

== Reporte 3. Lectores morosos (al menos un préstamo abierto y vencido) ==
SELECT l.documento, l.nombres, l.apellidos
FROM   lector l
WHERE  EXISTS (SELECT 1
               FROM   prestamo p
               WHERE  p.id_lector = l.id_lector
                 AND  p.fecha_devolucion IS NULL
                 AND  p.fecha_vencimiento < CURRENT_DATE)
ORDER BY l.apellidos;

== Reporte 4. Préstamos en mora (lista de notificación) ==
SELECT l.nombres || ' ' || l.apellidos          AS lector,
       l.correo, l.telefono,
       b.titulo,
       p.fecha_vencimiento,
       CURRENT_DATE - p.fecha_vencimiento        AS dias_mora
FROM   prestamo p
JOIN   lector   l ON l.id_lector   = p.id_lector
JOIN   ejemplar e ON e.id_ejemplar = p.id_ejemplar
JOIN   libro    b ON b.isbn        = e.isbn
WHERE  p.fecha_devolucion IS NULL
  AND  p.fecha_vencimiento < CURRENT_DATE
ORDER BY dias_mora DESC, lector;
